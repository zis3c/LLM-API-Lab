# Instant execution without build toolchain
curl -X POST https://api.openai.com/v1/chat/completions \
  -H "Authorization: Bearer $OPENAI_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "gpt-4o-mini",
    "messages": [{"role": "user", "content": "Hello World"}]
  }'
// config/providers.json routing registry
{
  "default": "openai",
  "providers": {
    "openai": { "env": "OPENAI_API_KEY", "baseUrl": "https://api.openai.com/v1" },
    "anthropic": { "env": "ANTHROPIC_API_KEY", "baseUrl": "https://api.anthropic.com" },
    "google": { "env": "GOOGLE_AI_API_KEY", "baseUrl": "https://generativelanguage.googleapis.com" }
  }
}
# Local environment configuration
APP_ENV=development
API_VERSION=v1
DEFAULT_PROVIDER=openai
DEFAULT_MODEL=gpt-4o-mini
LOG_LEVEL=info
